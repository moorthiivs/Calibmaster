const { HyperFormula } = require("hyperformula");

const generatePdfFromSheet = async (
  handsontableJson,
  MergedCells,
  Styles,
  selectedSheetInput,
  decimalPoint,
  Layout,
  fontSize
) => {
  try {
    const sheetKey =
      typeof selectedSheetInput === "object"
        ? selectedSheetInput.value
        : selectedSheetInput;

    const sheetData = handsontableJson[sheetKey] || [];
    const mergedCells = MergedCells?.[sheetKey] || [];
    //const stylesArray = Styles?.[sheetKey] || [];


    const rawStyleObj = Styles?.[sheetKey] || {};
    // const stylesArray = Object.values(rawStyleObj).filter(
    //   (s) => typeof s === "object" && s?.row !== undefined && s?.col !== undefined
    // );

    const stylesArray = Object.values(rawStyleObj).filter(
      (item) => typeof item === "object" && item?.row !== undefined && item?.col !== undefined
    );
    const conditionalStyles = rawStyleObj.conditionalFormatting || [];

    console.log(conditionalStyles, "conditionalStyles");

    let decimalPointArray = {};

    if (decimalPoint && typeof decimalPoint === "object") {
      const allPrecisions = {};

      for (const [sheetName, sheetPrecision] of Object.entries(decimalPoint)) {
        for (const [cellKey, value] of Object.entries(sheetPrecision)) {
          allPrecisions[cellKey] = value;
        }
      }

      decimalPointArray = allPrecisions;
    }

    if (!sheetData.length) return [];

    const hfInstance = HyperFormula.buildFromSheets(handsontableJson, {
      licenseKey: "gpl-v3",
    });

    const sheetId = hfInstance.getSheetId(sheetKey);
    if (sheetId === undefined) {
      throw new Error(`Sheet "${sheetKey}" not found`);
    }

    const evaluatedData = hfInstance.getSheetValues(sheetId);

    const maxCols = Math.max(...evaluatedData.map((row) => row.length), 0);
    const paddedData = evaluatedData.map((row) => {
      const newRow = Array(maxCols).fill("");
      return row.concat(newRow.slice(row.length));
    });

    const splitTables = [];
    let currentChunk = [];
    let startRowIndexes = [];

    paddedData.forEach((row, idx) => {
      const isEmptyRow = row.every((cell) => cell === "");
      if (isEmptyRow) {
        if (currentChunk.length) {
          splitTables.push(currentChunk);
          startRowIndexes.push(idx - currentChunk.length);
          currentChunk = [];
        }
      } else {
        currentChunk.push(row);
      }
    });

    if (currentChunk.length) {
      splitTables.push(currentChunk);
      startRowIndexes.push(paddedData.length - currentChunk.length);
    }

    const cleanMergeDefinitions = (
      mergedCells,
      tableData,
      startRow,
      endRow,
      maxCols
    ) => {
      return mergedCells
        .filter(
          ({ row, col, rowspan, colspan }) =>
            row >= startRow &&
            col >= 0 &&
            rowspan > 0 &&
            colspan > 0 &&
            row < endRow &&
            col < maxCols &&
            row + rowspan - 1 < endRow &&
            col + colspan - 1 < maxCols
        )
        .map(({ row, col, rowspan, colspan }) => ({
          row: row - startRow,
          col,
          rowspan: Math.min(rowspan, tableData.length - (row - startRow)),
          colspan: Math.min(colspan, maxCols - col),
        }))
        .filter(
          (cell, index, self) =>
            index ===
            self.findIndex((c) => c.row === cell.row && c.col === cell.col)
        );
    };

    // More accurate table height estimation
    const estimateTableHeight = (tableData, fontSize = 8, columnCount = 0) => {
      const baseRowHeight = fontSize * 1.5 + 4; // More conservative row height
      const headerHeight = (fontSize + 2) * 1.8 + 6; // Header padding
      const totalRows = tableData.length;

      // Minimal penalty for wide tables
      const wideTablePenalty = columnCount > 15 ? 1 : 0;
      const adjustedRowHeight = baseRowHeight + wideTablePenalty;

      return headerHeight + (totalRows - 1) * adjustedRowHeight + 5; // Minimal buffer
    };

    // More aggressive page fitting logic - only break when absolutely necessary
    const canFitOnPage = (tableHeight, currentPageHeight = 0, maxPageHeight = 750) => {
      const remainingSpace = maxPageHeight - currentPageHeight;
      const minimumRequiredSpace = tableHeight + 20; // Minimal buffer
      return remainingSpace >= minimumRequiredSpace;
    };

    // Calculate dynamic column widths to prevent overflow
    const calculateColumnWidths = (columnCount) => {
      const pageWidth = 520; // Available page width in points
      const minColumnWidth = 25; // Minimum column width
      const maxColumnWidth = 80; // Maximum column width

      if (columnCount <= 8) {
        // For normal tables, use equal distribution
        return Array(columnCount).fill("*");
      } else if (columnCount <= 15) {
        // For medium tables, use auto sizing
        return Array(columnCount).fill("auto");
      } else {
        // For very wide tables, calculate fixed widths
        const calculatedWidth = Math.max(minColumnWidth, Math.min(maxColumnWidth, pageWidth / columnCount));
        return Array(columnCount).fill(calculatedWidth);
      }
    };

    // Calculate optimal font size based on content
    const calculateFontSize = (columnCount, baseFontSize = 8) => {
      if (columnCount <= 8) return baseFontSize;
      if (columnCount <= 12) return Math.max(7, baseFontSize - 1);
      if (columnCount <= 18) return Math.max(6, baseFontSize - 2);
      return Math.max(5, baseFontSize - 3); // Minimum readable size
    };

    const processedTables = [];
    let currentPageHeight = 0;
    const maxPageHeight = 750; // Slightly more aggressive page height

    splitTables.forEach((tableData, chunkIndex) => {
      const startRow = startRowIndexes[chunkIndex];
      const endRow = startRow + tableData.length;

      // Apply styles to each cell
      const tableStyles = stylesArray
        .filter((s) => s.row >= startRow && s.row < endRow)
        .map((s) => ({
          ...s,
          row: s.row - startRow,
        }));


      const conditionStyles = conditionalStyles
        .filter((s) => s.row >= startRow && s.row < endRow)
        .map((s) => ({
          ...s,
          row: s.row - startRow,
        }));

      const tableMergedCells = cleanMergeDefinitions(
        mergedCells,
        tableData,
        startRow,
        endRow,
        maxCols
      );

      // Skip empty columns during PDF generation
      const nonEmptyColumnIndexes = [];
      for (let colIndex = 0; colIndex < maxCols; colIndex++) {
        const isNonEmptyColumn = tableData.some(
          (row) => row[colIndex] !== "" && row[colIndex] != null
        );
        if (isNonEmptyColumn) {
          nonEmptyColumnIndexes.push(colIndex);
        }
      }

      // Skip completely empty tables
      if (nonEmptyColumnIndexes.length === 0) {
        return;
      }

      const columnCount = nonEmptyColumnIndexes.length;
      const currentFontSize = calculateFontSize(columnCount, fontSize || 8);

      // Estimate table height with column count consideration
      const estimatedHeight = estimateTableHeight(tableData, currentFontSize, columnCount);

      // IMPROVED LOGIC: Only break page when data absolutely cannot fit
      let pageBreakBefore = false;

      // Check if this is not the first table AND it cannot fit on current page
      if (chunkIndex > 0 && currentPageHeight > 0 && !canFitOnPage(estimatedHeight, currentPageHeight, maxPageHeight)) {
        // Additional check: only break if the table has substantial content
        if (estimatedHeight > 100 || tableData.length > 3) {
          pageBreakBefore = true;
          currentPageHeight = 0; // Reset for new page
        }
      }

      // Update current page height
      currentPageHeight += estimatedHeight;

      // Ensure all cells in the table have valid objects
      const tableBody = tableData.map((row, rowIndex) =>
        nonEmptyColumnIndexes.map((colIndex) => {
          const cell = row[colIndex];
          const style = tableStyles.find(
            (s) => s.row === rowIndex && s.col === colIndex
          );

          const constyle = conditionStyles.find(
            (s) => s.row === rowIndex && s.col === colIndex
          );

          let formattedText = "";
          let numericValue = null;

          if (
            decimalPoint &&
            typeof decimalPoint === "object" &&
            decimalPoint[sheetKey]
          ) {
            const key = `${startRow + rowIndex}-${colIndex}`;
            const decimalPlaces = decimalPoint[sheetKey][key];

            if (decimalPlaces != null) {
              const parsed = parseFloat(cell);
              if (!isNaN(parsed)) {
                numericValue = parsed;
                formattedText = parsed.toFixed(decimalPlaces);
              }
            }
          }

          if (formattedText === "") {
            formattedText = (cell ?? "").toString();
          }

          const cellStyle = {
            text: formattedText,
            alignment: "center",
            bold: false,
            color: "#000000",
            fillColor: null,
            noWrap: columnCount > 10, // Only prevent wrapping for wide tables
          };



          if (constyle) {
            if (constyle.fontColor) cellStyle.color = constyle.fontColor;
            if (constyle.backgroundColor) cellStyle.fillColor = constyle.backgroundColor;
          }

          if (style) {
            if (style.fontColor) cellStyle.color = style.fontColor;
            if (style.backgroundColor)
              cellStyle.fillColor = style.backgroundColor;
            if (style.bold) cellStyle.bold = true;

            if (typeof style.className === "string") {
              if (style.className.includes("htLeft"))
                cellStyle.alignment = "left";
              if (style.className.includes("htRight"))
                cellStyle.alignment = "right";
              if (style.className.includes("htCenter"))
                cellStyle.alignment = "center";
            }
          }

          return cellStyle;
        })
      );

      // Apply merged cells
      tableMergedCells.forEach(({ row, col, rowspan, colspan }) => {
        if (
          row >= 0 &&
          col >= 0 &&
          row < tableBody.length &&
          col < tableBody[row].length &&
          row + rowspan <= tableBody.length &&
          col + colspan <= tableBody[row].length
        ) {
          tableBody[row][col].rowSpan = rowspan;
          tableBody[row][col].colSpan = colspan;

          // Mark merged cells (except the first one) as empty
          for (let r = row; r < row + rowspan; r++) {
            for (let c = col; c < col + colspan; c++) {
              if (r !== row || c !== col) {
                if (tableBody[r] && tableBody[r][c]) {
                  tableBody[r][c] = {};
                }
              }
            }
          }
        }
      });

      // Calculate optimal column widths to prevent overflow
      const widths = calculateColumnWidths(columnCount);

      // Adjust margins based on column count and page width
      
      const getMarginSettings = (columnCount) => {
        if (columnCount <= 8) return [0, 2, 0, 5];
        if (columnCount <= 12) return [10, 2, 10, 5];
        if (columnCount <= 18) return [20, 2, 20, 5];
        return [30, 2, 30, 5]; // Maximum compression for very wide tables
      };

      const marginSettings = getMarginSettings(columnCount);

      // Enhanced layout for different table sizes
      const getTableLayout = (columnCount) => {
        const baseLayout = Layout || {
          hLineWidth: function (i, node) {
            return 0.5;
          },
          vLineWidth: function (i, node) {
            return 0.5;
          },
          hLineColor: function (i, node) {
            return '#000000';
          },
          vLineColor: function (i, node) {
            return '#000000';
          },
        };

        // For very wide tables, use thinner lines
        if (columnCount > 15) {
          return {
            ...baseLayout,
            hLineWidth: function (i, node) {
              return 0.3;
            },
            vLineWidth: function (i, node) {
              return 0.3;
            },
          };
        }

        return baseLayout;
      };

      const tableConfig = {
        columns: [
          {
            width: "*",
            table: {
              headerRows: 1,
              widths,
              body: tableBody,
            },
            layout: getTableLayout(columnCount),
            fontSize: currentFontSize,
            style: "ninethTable",
            margin: marginSettings,
            alignment: "center",
          },
        ],
        alignment: "center",
        columnGap: 0,
        // Only add page break if absolutely necessary
        ...(pageBreakBefore && { pageBreak: "before" }),
        // Remove unbreakable for better flexibility
        // unbreakable: true,
      };

      processedTables.push(tableConfig);
    });

    return processedTables;
  } catch (err) {
    console.error("Error generating PDF:", err);
    return [
      {
        text: `Error generating table: ${err.message}`,
        color: "red",
        margin: [0, 10, 0, 10],
      },
    ];
  }
};

module.exports = { generatePdfFromSheet };
